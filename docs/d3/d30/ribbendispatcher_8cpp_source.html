34QJsonObject RibbenDispatcher::dispatch(const QJsonObject &request) const
35{
36 const QJsonValue id = request.value(QStringLiteral("id"));
37 const QString method = request.value(QStringLiteral("method")).toString();
38
39 if (method.isEmpty())
40 {
41 return ErrorResponse(id, RibbenError::InvalidRequest, QStringLiteral("Missing \"method\"."));
42 }
43
44 const QJsonValue paramsValue = request.value(QStringLiteral("params"));
45 const QJsonObject params = paramsValue.isObject() ? paramsValue.toObject() : QJsonObject();
46
47 try
48 {
49 return ResultResponse(id, invoke(method, params));
50 }
51 catch (const RibbenError &error)
52 {
53 return ErrorResponse(id, error.code(), error.message());
54 }
55}
