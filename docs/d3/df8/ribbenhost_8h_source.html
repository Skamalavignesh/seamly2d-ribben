40/**
41 * @brief Error thrown by a RibbenHost method to report a failed request
42 * (e.g. "no pattern is open", "unknown increment name"). RibbenDispatcher
43 * catches this and turns it into a JSON-RPC error response; it never
44 * crosses the socket boundary as a raw C++ exception.
45 */
46class RibbenError
47{
48public:
49 RibbenError(int code, const QString &message)
50 : m_code(code)
51 , m_message(message)
52 {}
53
54 int code() const { return m_code; }
55 QString message() const { return m_message; }
56
57 // Standard JSON-RPC 2.0 reserved codes, plus our own range (>= 1).
58 enum Code
59 {
60 InvalidRequest = -32600,
61 MethodNotFound = -32601,
62 InvalidParams = -32602,
63 InternalError = -32603,
64 Unauthorized = 1,
65 NoPatternOpen = 2,
66 NotFound = 3,
67 InvalidFormula = 4,
68 };
69
70private:
71 int m_code;
72 QString m_message;
73};

75/**
76 * @brief RibbenHost is what a live JSON-RPC request is actually dispatched
77 * against: a running Seamly2D window with a pattern open. Every method
78 * mirrors a tool already exposed by the file-based "seamly2d MCP" server,
79 * but operates on the live, in-memory document instead of a file on disk,
80 * so edits are reflected immediately in the open window (and the file-based
81 * server's read tools keep working once the user saves).
82 *
83 * Implementations run on the GUI thread (RibbenServer never hands work to a
84 * worker thread), so it's safe for a method to touch Qt widgets/doc classes
85 * directly, the same way a menu action's slot would.
86 */
87class RibbenHost
88{
89public:
90 virtual ~RibbenHost() = default;
91
92 /// Health check. Always succeeds if a host is reachable at all.
93 virtual QJsonObject ping() = 0;
94
95 /// Current file path, unsaved-changes flag, unit, and piece count.
96 virtual QJsonObject getStatus() = 0;
97
98 /// Description, notes, unit, and increment (custom variable) count.
99 /// Throws RibbenError(NoPatternOpen) if nothing is open.
100 virtual QJsonObject readPattern() = 0;
101
102 /// name/formula/value/description for every increment (custom variable)
103 /// in the open pattern. Throws RibbenError(NoPatternOpen) if nothing is
104 /// open.
105 virtual QJsonArray listIncrements() = 0;
106
107 /// Sets one increment's formula and rebuilds the pattern's geometry from
108 /// it (the same "set formula, then recompute" sequence the Variables
109 /// dialog performs on close). Throws RibbenError(NotFound) if no
110 /// increment has that name, or RibbenError(InvalidFormula) if the new
111 /// formula doesn't evaluate.
112 virtual QJsonObject updateIncrement(const QString &name, const QString &formula) = 0;
113
114 /// Replaces the pattern's free-text notes.
115 /// Throws RibbenError(NoPatternOpen) if nothing is open.
116 virtual QJsonObject setPatternNotes(const QString &text) = 0;
117
118 // -- Live geometry (draft points/lines) --
119 //
120 // Unlike everything above, these mutate the open pattern's actual draft
121 // geometry, not just its parametric/metadata layer, and the result
122 // appears on screen immediately -- the same "insert into the live XML
123 // document, then run the app's own full-rebuild-from-XML pass" sequence
124 // that already runs, live, every time the user hits Undo/Redo (see
125 // MainWindow::fullParseFile / doc->Parse(Document::FullParse)). New
126 // elements are only ever appended at the end of a draft block's
127 // <calculation>, so a referenced point always already exists by the
128 // time it's referenced -- dependency ordering takes care of itself.
129 //
130 // Covers the same handful of point types as the file-based
131 // xml_geometry.py tools in the sibling seamly2d-mcp project, kept in
132 // sync with it deliberately: "single" (an anchor point, no
133 // dependencies), "endLine" (length+angle from an existing point),
134 // "alongLine" (length along an existing line), and a plain connecting
135 // line. All throw RibbenError(NotFound) for an unknown draft block or
136 // referenced point, and RibbenError::InvalidParams for an invalid name.
137
138 /// Points in one draft block, in creation order, with their raw
139 /// attributes -- mirrors the file-based list_points tool.
140 virtual QJsonArray listPoints(const QString &draftBlockName) = 0;
141
142 /// Adds an anchor point at explicit (x, y) canvas coordinates (in the
143 /// pattern's own unit). The only point type with no dependencies.
144 virtual QJsonObject addPointSingle(const QString &draftBlockName, const QString &name, double x, double y) = 0;
145
146 /// Adds a point at a given length and angle (each a plain number or a
147 /// Seamly2D formula) from an existing point (named or referenced by id).
148 virtual QJsonObject addPointEndLine(
149 const QString &draftBlockName, const QString &name, const QString &basePoint,
150 const QString &length, const QString &angle, const QString &lineType) = 0;
151
152 /// Adds a point at a given length along the line from firstPoint toward
153 /// secondPoint.
154 virtual QJsonObject addPointAlongLine(
155 const QString &draftBlockName, const QString &name, const QString &firstPoint,
156 const QString &secondPoint, const QString &length) = 0;
157
158 /// Draws a plain visual line connecting two existing points.
159 virtual QJsonObject addLine(
160 const QString &draftBlockName, const QString &firstPoint, const QString &secondPoint,
161 const QString &lineType) = 0;
162
163 // -- Live curves, arcs, and piece outlines --
164 //
165 // Same idea as the point/line methods above, kept in sync with the
166 // sibling seamly2d-mcp project's file-based add_spline/add_arc/
167 // add_piece tools (same attribute shapes, same defaults). All throw
168 // RibbenError(NotFound) for an unknown draft block or referenced
169 // point/curve.
170
171 /// Draws a cubic-Bezier curve between two existing points (the "Curve"
172 /// tool), each end with its own angle+length tangent handle.
173 virtual QJsonObject addSpline(
174 const QString &draftBlockName, const QString &firstPoint, const QString &secondPoint,
175 const QString &angle1, const QString &length1, const QString &angle2, const QString &length2) = 0;
176
177 /// Draws a circular arc around an existing center point (the "Arc" tool).
178 virtual QJsonObject addArc(
179 const QString &draftBlockName, const QString &centerPoint, const QString &radius,
180 const QString &angle1, const QString &angle2) = 0;
181
182 /// Pieces (seam-allowance outlines) in one draft block, with their raw
183 /// attributes plus their outline as an array of node objects -- mirrors
184 /// the file-based list_pieces tool.
185 virtual QJsonArray listPieces(const QString &draftBlockName) = 0;
186
187 /// Creates a piece (seam-allowance outline) from existing points/
188 /// curves. `outline` is an ordered array of objects, each with exactly
189 /// one of "point"/"spline"/"arc" (a reference: point name/id, or a
190 /// spline/arc id) plus an optional "reverse" bool (ignored for point
191 /// nodes). Must have at least 2 entries. Each referenced point/spline/
192 /// arc is automatically promoted into the draft block's <modeling>
193 /// section first -- Seamly2D's own internal representation, mirrored
194 /// here -- so the caller never has to think about that layer.
195 /// Throws RibbenError(InvalidParams) for fewer than 2 nodes or an
196 /// ambiguous/missing point-spline-arc key in an entry.
197 virtual QJsonObject addPiece(
198 const QString &draftBlockName, const QString &name, const QJsonArray &outline,
199 bool seamAllowance, const QString &seamAllowanceWidth) = 0;
200};

202#endif // RIBBENHOST_H
