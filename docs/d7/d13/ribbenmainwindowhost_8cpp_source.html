125 void RibbenValidateName(const QString &name)
126 {
127 // Roughly the schema's "shortName" rules (src/libs/ifc/schema/pattern/*.xsd):
128 // no leading digit, no whitespace/formula-operator characters. Kept in
129 // sync with the equivalent check in the sibling seamly2d-mcp project's
130 // xml_geometry.py.
131 static const QRegularExpression invalidChars(QStringLiteral(R"([\s*/&|!<>^\-+.,=?:;'"])"));
132 if (name.isEmpty())
133 {
134 throw RibbenError(RibbenError::InvalidParams, QObject::tr("name must not be empty"));
135 }
136 if (name.at(0).isDigit())
137 {
138 throw RibbenError(RibbenError::InvalidParams,
139 QObject::tr("invalid name \"%1\": must not start with a digit").arg(name));
140 }
141 if (invalidChars.match(name).hasMatch())
142 {
143 throw RibbenError(RibbenError::InvalidParams,
144 QObject::tr("invalid name \"%1\": must not contain whitespace or "
145 "formula-operator characters").arg(name));
146 }
147 }

237//---------------------------------------------------------------------------------------------------------------------
238/// @brief toggleRibbenAddon starts or stops the embedded live server and
239/// persists the new on/off state so it's restored next launch.
240/// @param checked true to start listening, false to stop.
241//---------------------------------------------------------------------------------------------------------------------
242void MainWindow::toggleRibbenAddon(bool checked)
243{
244 RibbenSettings settings;
245 settings.setEnabled(checked);
246
247 if (!checked)
248 {
249 if (m_ribbenServer != nullptr)
250 {
251 m_ribbenServer->stop();
252 }
253 setStatusMessage(tr("Ribben addon stopped."));
254 return;
255 }
256
257 if (m_ribbenServer == nullptr)
258 {
259 m_ribbenServer = new RibbenServer(this, this);
260 connect(m_ribbenServer, &RibbenServer::connectionRejected, this, [this](const QString &address)
261 {
262 setStatusMessage(tr("Ribben addon: rejected connection from %1 (not in the allow-list).").arg(address));
263 });
264 }
265
266 QString errorMessage;
267 if (!m_ribbenServer->start(&errorMessage))
268 {
269 m_ribbenAction->blockSignals(true);
270 m_ribbenAction->setChecked(false);
271 m_ribbenAction->blockSignals(false);
272 settings.setEnabled(false);
273
274 QMessageBox::warning(this, tr("Ribben addon"),
275 tr("Could not start the live MCP server: %1").arg(errorMessage));
276 return;
277 }
278
279 setStatusMessage(tr("Ribben addon listening on 127.0.0.1:%1.").arg(settings.port()));
280}

317QJsonArray MainWindow::listIncrements()
318{
319 QJsonArray array;
320
321 const QMap<QString, QSharedPointer<CustomVariable>> variables = pattern->variablesData();
322 for (auto it = variables.constBegin(); it != variables.constEnd(); ++it)
323 {
324 // A plain "const CustomVariable *" (not the smart pointer's own
325 // constness) is what actually picks the "qreal GetValue() const"
326 // overload below over the mutable "qreal *GetValue()" one --
327 // QSharedPointer::operator->() always yields a non-const pointee.
328 const CustomVariable *variable = it.value().data();
329
330 QJsonObject entry;
331 entry.insert(QStringLiteral("name"), it.key());
332 entry.insert(QStringLiteral("formula"), variable->GetFormula());
333 entry.insert(QStringLiteral("value"), variable->GetValue());
334 entry.insert(QStringLiteral("description"), variable->GetDescription());
335 entry.insert(QStringLiteral("valid"), variable->IsFormulaOk());
336 array.append(entry);
337 }
338
339 return array;
340}

343/// @brief updateIncrement sets one increment's formula and rebuilds the
344/// pattern's geometry from it -- the same "commit formula, then recompute"
345/// sequence DialogVariables performs when it closes (see
346/// DialogVariables::refreshPattern), just without needing the dialog open.
347//---------------------------------------------------------------------------------------------------------------------
348QJsonObject MainWindow::updateIncrement(const QString &name, const QString &formula)
349{
350 try
351 {
352 pattern->getVariable<CustomVariable>(name); // throws VExceptionBadId if unknown
353 }
354 catch (const VExceptionBadId &error)
355 {
356 Q_UNUSED(error)
357 throw RibbenError(RibbenError::NotFound, tr("No increment named \"%1\".").arg(name));
358 }
359
360 doc->setVariableFormula(name, formula);
361
362 try
363 {
364 doc->LiteParseTree(Document::LiteParse);
365 }
366 catch (const VException &error)
367 {
368 throw RibbenError(RibbenError::InvalidFormula, error.ErrorMessage());
369 }

372 throw RibbenError(RibbenError::InvalidFormula, error.GetMsg());
373 }
374
375 // See listIncrements() for why this needs to be a "const CustomVariable *"
376 // rather than the (non-const-pointee) QSharedPointer itself.
377 const CustomVariable *updated = pattern->getVariable<CustomVariable>(name).data();
378
379 QJsonObject result;
380 result.insert(QStringLiteral("name"), name);
381 result.insert(QStringLiteral("formula"), updated->GetFormula());
382 result.insert(QStringLiteral("value"), updated->GetValue());
383 result.insert(QStringLiteral("valid"), updated->IsFormulaOk());

398/// @brief ribbenReparseOrRollback re-derives the whole live scene from doc's
399/// XML -- the same doc->Parse(Document::FullParse) call MainWindow::fullParseFile()
400/// makes on every Undo/Redo, so a new point genuinely appears on screen
401/// immediately, the same way an undone/redone one would. Called directly
402/// rather than through fullParseFile() because that slot *swallows*
403/// VExceptionObjectError/VExceptionConversionError internally (logs them and
404/// disables the GUI) instead of propagating them -- which would leave a bad
405/// live_add_point_* call invisible to the caller and the GUI silently
406/// disabled. On failure here, the just-inserted element is removed and the
407/// document re-parsed again to restore the prior good state before the
408/// error is reported.
409//---------------------------------------------------------------------------------------------------------------------
410void MainWindow::ribbenReparseOrRollback(QDomElement &parent, QDomElement &child)
411{
412 try
413 {
414 toolProperties->clearPropertyBrowser();
415 setGuiEnabled(true);
416 doc->Parse(Document::FullParse);
417 }
418 catch (const VException &error)
419 {
420 parent.removeChild(child);
421 try
422 {
423 doc->Parse(Document::FullParse);
424 }
425 catch (const VException &)
426 {
427 // Best-effort restore. If this also fails, the live document was
428 // already broken before our change; there's nothing more we can
429 // safely do from here.
430 }
431 throw RibbenError(RibbenError::InvalidFormula, error.ErrorMessage());
432 }
433}

436QJsonArray MainWindow::listPoints(const QString &draftBlockName)
437{
438 const QDomElement draftBlock = RibbenRequireDraftBlock(doc, draftBlockName);
439 const QDomElement calculation = RibbenRequireCalculation(draftBlock);
440
441 QJsonArray result;
442 QDomElement point = calculation.firstChildElement(VAbstractPattern::TagPoint);
443 while (!point.isNull())
444 {
445 QJsonObject entry;
446 const QDomNamedNodeMap attrs = point.attributes();
447 for (int i = 0; i < attrs.count(); ++i)
448 {
449 const QDomAttr attr = attrs.item(i).toAttr();
450 entry.insert(attr.name(), attr.value());
451 }
452 result.append(entry);
453 point = point.nextSiblingElement(VAbstractPattern::TagPoint);
454 }
455 return result;
456}

460/// -- the only point type with no dependencies -- and makes it appear on

669 const QDomAttr attr = attrs.item(i).toAttr();
670 entry.insert(attr.name(), attr.value());
671 }
672

674 const QDomElement nodesEl = piece.firstChildElement(VAbstractPattern::TagNodes);
675 QDomElement nodeEl = nodesEl.firstChildElement(VAbstractPattern::TagNode);

680 for (int i = 0; i < nodeAttrs.count(); ++i)
681 {
682 const QDomAttr attr = nodeAttrs.item(i).toAttr();
683 node.insert(attr.name(), attr.value());
684 }
685 outline.append(node);
686 nodeEl = nodeEl.nextSiblingElement(VAbstractPattern::TagNode);
687 }
688 entry.insert(QStringLiteral("outline"), outline);
689
690 result.append(entry);
691 piece = piece.nextSiblingElement(VAbstractPattern::TagPiece);
692 }

697/// @brief addPiece builds a seam-allowance outline from existing points/
698/// curves, live. Mirrors the file-based add_piece tool: every referenced
699/// point/spline/arc first gets promoted into <modeling> (Seamly2D can't
700/// have a <piece> reference <calculation> geometry directly), then the
701/// piece's <nodes> reference those modeling ids. All references are
702/// resolved and validated up front, before anything is mutated, so a bad
703/// reference never leaves a partial piece behind; if the final reparse
704/// still fails (e.g. a curve whose formula doesn't evaluate), every element
705/// this call inserted -- all the modeling wrappers and the piece itself --
706/// is rolled back together, the same rollback guarantee
707/// ribbenReparseOrRollback gives a single-element insert.
708//---------------------------------------------------------------------------------------------------------------------
709QJsonObject MainWindow::addPiece(
710 const QString &draftBlockName, const QString &name, const QJsonArray &outline,
711 bool seamAllowance, const QString &seamAllowanceWidth)
712{
713 if (outline.size() < 2)
714 {
715 throw RibbenError(RibbenError::InvalidParams, QObject::tr("a piece outline needs at least 2 nodes"));
716 }
717
718 QDomElement draftBlock = RibbenRequireDraftBlock(doc, draftBlockName);
719 QDomElement modeling = RibbenRequireModeling(draftBlock);
720 QDomElement pieces = RibbenRequirePieces(draftBlock);
721
722 QList<RibbenPieceNode> resolvedNodes;
723 resolvedNodes.reserve(outline.size());
724 for (const QJsonValue &entryValue : outline)
725 {
726 const QJsonObject entry = entryValue.toObject();
727 const bool hasPoint = entry.contains(QStringLiteral("point"));
728 const bool hasSpline = entry.contains(QStringLiteral("spline"));
729 const bool hasArc = entry.contains(QStringLiteral("arc"));
730 if (int(hasPoint) + int(hasSpline) + int(hasArc) != 1)
731 {
732 throw RibbenError(RibbenError::InvalidParams,
733 QObject::tr("each outline entry must have exactly one of point/spline/arc"));
734 }
735
736 RibbenPieceNode node;
737 node.reverse = entry.value(QStringLiteral("reverse")).toBool(false);
738
739 if (hasPoint)
740 {
741 const QDomElement calcEl = RibbenFindPointByRef(doc, entry.value(QStringLiteral("point")).toString());
742 node.nodeType = VAbstractPattern::NodePoint;
743 node.modelingType = RIBBEN_MODELING_TYPE_PLAIN;
744 node.modelingTag = VAbstractPattern::TagPoint;
745 node.calcId = calcEl.attribute(VDomDocument::AttrId);
746 }
747 else if (hasSpline)
748 {
749 const QDomElement calcEl = RibbenFindElementByTagAndId(
750 doc, VAbstractPattern::TagSpline, entry.value(QStringLiteral("spline")).toString());
751 node.nodeType = VAbstractPattern::NodeSpline;
752 node.modelingType = RIBBEN_MODELING_TYPE_SPLINE;
753 node.modelingTag = VAbstractPattern::TagSpline;
754 node.calcId = calcEl.attribute(VDomDocument::AttrId);
755 }
756 else
757 {
758 const QDomElement calcEl = RibbenFindElementByTagAndId(
759 doc, VAbstractPattern::TagArc, entry.value(QStringLiteral("arc")).toString());
760 node.nodeType = VAbstractPattern::NodeArc;
761 node.modelingType = RIBBEN_MODELING_TYPE_PLAIN;
762 node.modelingTag = VAbstractPattern::TagArc;
763 node.calcId = calcEl.attribute(VDomDocument::AttrId);
764 }
765 resolvedNodes.append(node);
766 }
767
768 // Every reference validated -- now mutate. Track every inserted element
769 // (modeling wrappers + the piece) so a failed reparse can roll all of
770 // them back together.
771 QDomElement nodesEl = doc->createElement(VAbstractPattern::TagNodes);
772 QList<QDomElement> insertedModelingElements;
773
774 for (const RibbenPieceNode &node : resolvedNodes)
775 {
776 const quint32 modelingId = VContainer::getNextId();
777 QDomElement modelingEl = doc->createElement(node.modelingTag);
778 doc->SetAttribute(modelingEl, VDomDocument::AttrId, modelingId);
779 doc->SetAttribute(modelingEl, AttrType, node.modelingType);
780 doc->SetAttribute(modelingEl, AttrIdObject, node.calcId);
781 modeling.appendChild(modelingEl);
782 insertedModelingElements.append(modelingEl);
783
784 QDomElement nodeEl = doc->createElement(VAbstractPattern::TagNode);
785 doc->SetAttribute(nodeEl, AttrIdObject, modelingId);
786 doc->SetAttribute(nodeEl, AttrType, node.nodeType);
787 if (node.nodeType != VAbstractPattern::NodePoint)
788 {
789 doc->SetAttribute(nodeEl, VAbstractPattern::AttrNodeReverse, node.reverse ? 1 : 0);
790 }
791 nodesEl.appendChild(nodeEl);
792 }
793
794 const quint32 pieceId = VContainer::getNextId();
795 QDomElement piece = doc->createElement(VAbstractPattern::TagPiece);
796 doc->SetAttribute(piece, VDomDocument::AttrId, pieceId);
797 doc->SetAttribute(piece, AttrName, name);
798 doc->SetAttribute(piece, RIBBEN_ATTR_PIECE_VERSION, 2);
799 doc->SetAttribute(piece, AttrClosed, 1);
800 doc->SetAttribute(piece, RIBBEN_ATTR_SEAM_ALLOWANCE, seamAllowance);
801 if (seamAllowance)
802 {
803 doc->SetAttribute(piece, VAbstractPattern::AttrWidth, seamAllowanceWidth);
804 }
805 piece.appendChild(nodesEl);
806 pieces.appendChild(piece);
807
808 try
809 {
810 toolProperties->clearPropertyBrowser();
811 setGuiEnabled(true);
812 doc->Parse(Document::FullParse);
813 }
814 catch (const VException &error)
815 {
816 pieces.removeChild(piece);
817 for (QDomElement modelingEl : insertedModelingElements)
818 {
819 modeling.removeChild(modelingEl);
820 }
821 try
822 {
823 doc->Parse(Document::FullParse);
824 }
825 catch (const VException &)
826 {
827 // Best-effort restore, same as ribbenReparseOrRollback.
828 }
829 throw RibbenError(RibbenError::InvalidFormula, error.ErrorMessage());
830 }
831
832 QJsonObject result;
833 result.insert(QStringLiteral("id"), static_cast<int>(pieceId));
834 return result;
835}
