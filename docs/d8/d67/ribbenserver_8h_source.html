2// @file ribbenserver.h
3//
4// @brief
5// Ribben addon: the embedded live JSON-RPC server. One newline-delimited
6// JSON object per request/response, e.g.
7//
8// {"id":1,"method":"ping","token":"<token>"}\n
9// {"jsonrpc":"2.0","id":1,"result":{"ok":true}}\n
10//
11// Runs entirely on the thread it was created on (the GUI thread, in
12// practice) via Qt's own event loop -- there is no worker thread and no
13// blocking accept() loop, so every RibbenHost method below runs exactly
14// like a menu action's slot would. Bound to loopback only in this version;
15// the allow-list (RibbenIpFilter) is defense in depth on top of that, and
16// every request must additionally present the token from RibbenSettings.
